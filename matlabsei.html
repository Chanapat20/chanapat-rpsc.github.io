Code: SEI simulation using AIPFM


clc
clear
close all
%Domain Spacing
nx = 200;
ny = 20;
L = 100*10^(-9);
W = 500*10^(-10);
nt = 9e4; %1e8 for detail, 6e7 for quick shot
T = 1e-5;
x = linspace(0,L,nx);
y = linspace(0,W,ny);
t = linspace(0,T,nt);
dx = x(2)-x(1);
dy = y(2)-y(1);
dt = t(2)-t(1);
nprint = 1000;
%Parameters
R = 8.314; %Gas Constant
Temp = 298; %Kelvin
G0_1 = -2.99;
G0_2 = -0.4;
G0_3 = -3.39;
kappa = 7.5*10^(-10);
w = 4 * 10^(9);
%k1 = 1.90e10; original (The scale are to large)
%k2 = 8.76e9; original (The scale are to large)
%k3 = 5.14; original (The scale are to large)
k1 = 1.90e5; %(Rescale)
k2 = 8.76e4; %(Rescale)
k3 = 5.14e-5; %(Rescale)
F = 96485.33289; %Faraday Constant
PSI0_1 = -2.18;
PSI0_2 = -0.21;
PSI0_3 = 0.65;
PSITerm = -3.04; %PSIe - PSIsol = -3.04 V
tol_dGr = 1e-3; % J/mol threshold for equilibrium condition
me = 9.1093837e-31; %Electron mass Kg
hbar = 1.054571e-34; %Reduced Planck Constant
dE_S2 = 0.24*1.6e-19; %Electron Barrier for S2 Joule
dE_S1 = 1.78*1.6e-19; %Electron Barrier for S1 Joule
beta_S1 = sqrt(2*me*dE_S1)/hbar; % tunneling decay constant (1/m)
beta_S2 = sqrt(2*me*dE_S2)/hbar; % lower = weaker decay (organic)
ae_lim  = 1e-8; % cutoff activity to prevent shrinkage
%Activity of species
aLi = 1;
aEC = 15;
aS1 = 1;
aS2 = 1;
% *activity* Important
ae = 1; %For case I from journal    
%Stoichiometric coefficients
vS1_2 = 1;
vS1_3 = 1;
vS2_1 = 0.5;
vS2_2 = 0.5;
vLi_1 = 1;
vLi_2 = 1;
vLi_3 = 2;
ve_1 = 1;
ve_2 = 1;
ve_3 = 2;
vEC_1 = 1;
vEC_3 = 1;
h = @(p) 6.*p.^5 - 15.*p.^4 + 10.*p.^3;
hprime = @(p) 30.*p.^4 - 60.*p.^3 + 30.*p.^2;
gwell = @(phiE,phiS1,phiS2) -w*(2-phiE.^2).*phiE.^2 - w*(2-phiS1.^2).*phiS1.^2 - w*(2-phiS2.^2).*phiS2.^2 + w*( phiE.^2.*phiS1.^2 + phiE.^2.*phiS2.^2 + phiS1.^2.*phiS2.^2 );
dgE = @(phiE,phiS1,phiS2) -4*w*phiE.*(1-phiE).*(1-2*phiE) + 2*w*phiE.*(phiS1.^2 + phiS2.^2);
dgS1 = @(phiE,phiS1,phiS2) -4*w*phiS1.*(1-phiS1).*(1-2*phiS1) + 2*w*phiS1.*(phiE.^2 + phiS2.^2);
dgS2 = @(phiE,phiS1,phiS2) -4*w*phiS2.*(1-phiS2).*(1-2*phiS2) + 2*w*phiS2.*(phiE.^2 + phiS1.^2);
%Adding phi
phi_S1 = zeros(ny,nx); %Inorganic
phi_S2 = zeros(ny,nx); %Organic
% Add 0.5nm Li2CO3 (S1) layer (1 grid point, since dx=0.5nm) [0.5nm is from Journal]
phi_S1(:, 1) = 1.0;
% Add 6nm Li2BDC (S2) layer (12 grid points, since 6nm/0.5nm = 12) [6nm from Journal]
% This layer starts after S1, from column 2 to 13
phi_S2(:, 2:13) = 1.0;
phi_E = 1 - phi_S1 - phi_S2; % both 0 = Electrolyte
%Interface Mobilities (Source ChatGPT)
LE = 1e-3;
LS1 = 1e-3;
LS2 = 1e-3;
%Laplacian term (Discretized) (2D for maybe future 2D growth)
lap = @(A) (([A(:,2:end) A(:,end)] - 2.*A + [A(:,1) A(:,1:end-1)])/(dx^2) + ([A(2:end,:); A(end,:)] - 2.*A + [A(1,:); A(1:end-1,:)])/(dy^2));
%Time stepping 
figure; % Create a new figure window
for n = 1:nt
    %Enforce physical bounds
    phi_E  = max(min(1 - phi_S1 - phi_S2,1),0);
    phi_S1 = max(min(phi_S1,1),0);
    phi_S2 = max(min(phi_S2,1),0);

    %Compute local SEI thickness
    d_SEI = sum((phi_S1 + phi_S2), 2) * dx;  % [m] per y-row

    %Electron activity from tunneling 
    ae_S1 = exp(-2 * beta_S1 .* d_SEI);
    ae_S2 = exp(-2 * beta_S2 .* d_SEI);
    ae = (phi_S1 .* ae_S1 + phi_S2 .* ae_S2 + phi_E * 1.0); % electrolyte ae ≈ 1
    ae = max(ae, ae_lim);
    
    %Update Gibbs free energies
    dGr1 = G0_1  + F*(PSITerm - PSI0_1) - R*Temp*log(((aLi^vLi_1)*(aEC^vEC_1)*(ae.^ve_1)) ./ (aS2.^vS2_1));
    dGr2 = G0_2  + F*(PSITerm - PSI0_2) - R*Temp*log(((aLi^vLi_2)*(aS2.^vS2_2)*(ae.^ve_2)) ./ (aS1.^vS1_2));
    dGr3 = G0_3  + F*(PSITerm - PSI0_3) - R*Temp*log(((aLi^vLi_3)*(aEC^vEC_3)*(ae.^ve_3)) ./ (aS1.^vS1_3));
    
    %Reaction rates (decrease as ae→ae_lim)
    R1 = -k1*(dGr1/(R*Temp));
    R2 = -k2*(dGr2/(R*Temp));
    R3 = -k3*(dGr3/(R*Temp));
    
    %Stop reactions when ΔG ≈ 0 (equilibrium reached)
    R1(ae <= ae_lim) = 0;
    R2(ae <= ae_lim) = 0;
    R3(ae <= ae_lim) = 0;

    if all(abs(R1(:)) < 1e-6 & abs(R2(:)) < 1e-6 & abs(R3(:)) < 1e-6)
    fprintf('Equilibrium reached at step %d, time = %.2e s\n', n, t(n));
    break;
    end
    %Source terms 
    src_S1 = hprime(phi_S1).*(R2 + R3);
    src_S2 = hprime(phi_S2).*(R1 - R2); %?? Originally R1 - R2(Not working) => R2 - R1 (Work) {Math :P}

    %AllenCahn updates
    phi_S1 = phi_S1 + dt*(-LS1*(dgS1(phi_E,phi_S1,phi_S2) - kappa*lap(phi_S1)) + src_S1);
    phi_S2 = phi_S2 + dt*(-LS2*(dgS2(phi_E,phi_S1,phi_S2) - kappa*lap(phi_S2)) + src_S2);

    %Re-enforce bounds and recalc electrolyte
    phi_S1 = max(min(phi_S1,1),0);
    phi_S2 = max(min(phi_S2,1),0);
    phi_E  = max(0, 1 - phi_S1 - phi_S2);
    % Visualization
    if mod(n, nprint) == 0
        phase_map = zeros(ny, nx);
        phase_map(phi_S1 > 0.5) = 1;
        phase_map(phi_S2 > 0.5) = 2;
        imagesc(x*1e9, y*1e9, phase_map);
        axis image;
        my_colors = [0 0 1; 1 0 0; 0 1 0];
        colormap(my_colors);
        colorbar('Ticks', [0.33, 1, 1.67], ...
                 'TickLabels', {'Electrolyte', 'S1 (Inorganic)', 'S2 (Organic)'});
        clim([0 2]);
        title(['SEI Phase Evolution at t = ', num2str(t(n), '%.2e'), ' s']);
        xlabel('x (nm)'); ylabel('y (nm)');
        drawnow;
    end
end
